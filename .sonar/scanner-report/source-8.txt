name: CI Pipeline

on:
  push:
    branches:
      - main
  pull_request:

jobs:
  build:
    runs-on: ubuntu-latest

    steps:
      - uses: actions/checkout@v3

      - name: Set up Python 3.12
        uses: actions/setup-python@v4
        with:
          python-version: '3.12'

      - name: Check for symbolic links
        run: |
          echo "Finding non-symlink files inside python-project-50..."
          find . ! -type l
      - name: Remove virtual environments
        run: |
          echo "Removing virtual environments..."
          rm -rf env .venv venv
      - name: Create virtual environment
        run: |
          python -m venv .venv
      - name: Install dependencies from pyproject.toml and tools
        run: |
          source .venv/bin/activate
          pip install --upgrade pip
          pip install .  # Установка зависимостей проекта
          pip install ruff pytest pytest-cov  # Установка инструментов
      - name: Build project
        run: |
          source .venv/bin/activate
          pip install --upgrade build
          python -m build
      - name: Remove old wheel files, keep only the latest
        run: |
          ls -t dist/*.whl | tail -n +2 | xargs -r rm --
      - name: Install the latest wheel
        run: |
          source .venv/bin/activate
          pip install dist/*.whl
      - name: Run lint via make
        run: |
          source .venv/bin/activate
          make lint
      - name: Run tests with coverage
        run: |
          source .venv/bin/activate
          pytest --cov=your_package_name --cov-report=xml
      # Удалены шаги с SonarQube
